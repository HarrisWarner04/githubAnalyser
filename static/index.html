Use this server directly inside Claude Desktop, Cursor, or Kiro IDE

Connect your remote MCP client to your Azure VM streamable HTTP endpoint:

{
  "mcpServers": {
    "github-analyzer": {
      "url": "http://YOUR_SERVER_HOST:8000/mcp",
      "transport": "http"
    }
  }
}

Add to claude_desktop_config.json:

{
  "mcpServers": {
    "github-analyzer": {
      "command": "python",
      "args": ["/path/to/github-analyzer-mcp/server.py"],
      "env": {
        "GEMINI_API_KEY": "your_gemini_key",
        "GROQ_API_KEY": "your_groq_key",
        "GEMINI_MODEL": "gemini-3.5-flash",
        "GROQ_MODEL": "llama-3.1-8b-instant"
      }
    }
  }
}
